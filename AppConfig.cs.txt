#nullable enable
// =========================================================================
// アプリケーション設定モデル / 設定ファイル読み込み
//
// appsettings.json から各クライアント（ClientA〜D, ProxyClientA/B）の
// 接続先設定と test3Func 用の ViewList 選択肢を読み込む。
// =========================================================================
using System;
using System.Collections.Generic;
using System.IO;
using System.Windows.Forms;
using Newtonsoft.Json;

namespace SocketApp.Models
{
    /// <summary>1クライアント分の接続先設定</summary>
    public class ClientConfig
    {
        /// <summary>接続先ホスト名または IP アドレス</summary>
        public string Host { get; set; } = "127.0.0.1";

        /// <summary>接続先ポート番号</summary>
        public int Port { get; set; } = 8080;
    }

    /// <summary>
    /// アプリ全体の設定。appsettings.json のルートオブジェクトに対応する。
    /// </summary>
    public class AppSettings
    {
        /// <summary>ClientA（自動運転コマンド送信用）の接続設定</summary>
        public ClientConfig ClientA { get; set; } = new ClientConfig();

        /// <summary>ClientB（test1〜3 送信用）の接続設定</summary>
        public ClientConfig ClientB { get; set; } = new ClientConfig();

        /// <summary>ClientC（エラー通知受信・サーバ起点リクエスト応答用）の接続設定</summary>
        public ClientConfig ClientC { get; set; } = new ClientConfig();

        /// <summary>ClientD（test4〜5 送信用）の接続設定</summary>
        public ClientConfig ClientD { get; set; } = new ClientConfig();

        /// <summary>ProxyServerA（TCPサーバとして動作）のポート設定。Host は未使用。</summary>
        public ClientConfig ProxyServerA { get; set; } = new ClientConfig();

        /// <summary>ProxyClientB（実機サーバ接続用）の接続設定</summary>
        public ClientConfig ProxyClientB { get; set; } = new ClientConfig();

        /// <summary>test3Func の viewList パラメータ選択肢リスト</summary>
        public List<string> ViewList { get; set; } = new List<string>();
    }

    /// <summary>appsettings.json の読み込みを担当する静的クラス</summary>
    public static class AppConfig
    {
        private const string ConfigFileName = "appsettings.json";

        public static AppSettings Load()
        {
            try
            {
                string path = Path.Combine(AppDomain.CurrentDomain.BaseDirectory, ConfigFileName);
                if (!File.Exists(path))
                {
                    var defaults = CreateDefaults();
                    File.WriteAllText(path, JsonConvert.SerializeObject(defaults, Formatting.Indented));
                    return defaults;
                }
                string json = File.ReadAllText(path);
                return JsonConvert.DeserializeObject<AppSettings>(json) ?? new AppSettings();
            }
            catch (Exception ex)
            {
                MessageBox.Show(
                    string.Format("設定ファイルの読み込みに失敗しました。\n{0}", ex.Message),
                    "設定エラー", MessageBoxButtons.OK, MessageBoxIcon.Warning);
                return new AppSettings();
            }
        }

        private static AppSettings CreateDefaults()
        {
            return new AppSettings
            {
                ClientA      = new ClientConfig { Host = "127.0.0.1", Port = 8001 },
                ClientB      = new ClientConfig { Host = "127.0.0.1", Port = 8002 },
                ClientC      = new ClientConfig { Host = "127.0.0.1", Port = 8003 },
                ClientD      = new ClientConfig { Host = "127.0.0.1", Port = 8004 },
                ProxyServerA = new ClientConfig { Host = "0.0.0.0", Port = 9001 },
                ProxyClientB = new ClientConfig { Host = "192.168.1.100", Port = 9002 },
                ViewList     = new List<string> { "view1", "view2", "view3" }
            };
        }
    }
}
