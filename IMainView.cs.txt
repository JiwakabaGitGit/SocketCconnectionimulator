#nullable enable
// =========================================================================
// IMainView - View インターフェース
//
// Presenter が View を操作するための契約。
// ClientA〜D（MockServer接続用）+ ProxyClientA/B（実機サーバ接続用）の
// 6クライアント分の接続管理 UI を定義する。
// =========================================================================
using System;
using System.Collections.Generic;

namespace SocketApp.Views
{
    public interface IMainView
    {
        // ---- 接続・切断ボタン（クライアント個別） ----
        bool ConnectAButtonEnabled { set; }
        bool DisconnectAButtonEnabled { set; }
        bool ConnectBButtonEnabled { set; }
        bool DisconnectBButtonEnabled { set; }
        bool ConnectCButtonEnabled { set; }
        bool DisconnectCButtonEnabled { set; }
        bool ConnectDButtonEnabled { set; }
        bool DisconnectDButtonEnabled { set; }
        // ---- ProxyServerA（TCPサーバ）起動/停止 ----
        bool StartProxyAButtonEnabled { set; }
        bool StopProxyAButtonEnabled { set; }
        bool SendProxyAButtonEnabled { set; }
        string ProxyASendText { get; }
        bool ConnectProxyBButtonEnabled { set; }
        bool DisconnectProxyBButtonEnabled { set; }

        // ---- その他ボタン ----
        bool StartDriveButtonEnabled { set; }
        bool EndDriveButtonEnabled { set; }
        bool Test1ButtonEnabled { set; }
        bool Test2ButtonEnabled { set; }
        bool Test3ButtonEnabled { set; }
        bool Test4ButtonEnabled { set; }
        bool Test5ButtonEnabled { set; }
        bool ClientCSendButtonEnabled { set; }
        bool ScenarioRunButtonEnabled { set; }
        bool ScenarioCancelButtonEnabled { set; }

        // ---- ステータスランプ ----
        void SetConnectionStatus(string clientName, bool isConnected);

        // ---- シナリオ状態表示 ----
        void SetScenarioStatus(string statusText);

        // ---- コンボボックス ----
        void SetViewListItems(IList<string> items);
        string SelectedViewListItem { get; }

        // ---- test4Func params 入力値 ----
        bool Test4Info1 { get; }
        bool Test4Info2 { get; }
        bool? Test4Info3 { get; }
        bool? Test4Info4 { get; }

        // ---- ログ出力 ----
        void AppendLog(string message, LogLevel level);
        bool FileLogEnabled { get; }

        // ---- ProxyClientB 受信履歴表 ----
        void AddProxyBHistoryRow(string time, string method, string id);

        // ---- 入力値取得 ----
        string ClientCMethod { get; }
        string ScenarioFilePath { get; }
        string ClientCResponseFilePath { get; }

        // ---- ダイアログ ----
        void ShowWarning(string message, string title);
        void ShowError(string message, string title);
        string? ShowOpenFileDialog(string title, string filter);

        // ---- イベント（View → Presenter）: 接続・切断 ----
        event EventHandler? ConnectAClicked;
        event EventHandler? DisconnectAClicked;
        event EventHandler? ConnectBClicked;
        event EventHandler? DisconnectBClicked;
        event EventHandler? ConnectCClicked;
        event EventHandler? DisconnectCClicked;
        event EventHandler? ConnectDClicked;
        event EventHandler? DisconnectDClicked;
        event EventHandler? StartProxyAClicked;
        event EventHandler? StopProxyAClicked;
        event EventHandler? SendProxyAClicked;
        event EventHandler? ConnectProxyBClicked;
        event EventHandler? DisconnectProxyBClicked;

        // ---- イベント（View → Presenter）: コマンド ----
        event EventHandler? StartDriveClicked;
        event EventHandler? EndDriveClicked;
        event EventHandler? Test1Clicked;
        event EventHandler? Test2Clicked;
        event EventHandler? Test3Clicked;
        event EventHandler? Test4Clicked;
        event EventHandler? Test5Clicked;
        event EventHandler? ClientCSendClicked;
        event EventHandler? ClearLogClicked;
        event EventHandler? ScenarioRunClicked;
        event EventHandler? ScenarioCancelClicked;
        event EventHandler? ScenarioBrowseClicked;
        event EventHandler? ClientCResponseBrowseClicked;
        event EventHandler? FormClosingEvent;

        // ---- イベント（View → Presenter）: JSON ファイル送信 ----
        event EventHandler? SendJsonAClicked;
        event EventHandler? SendJsonBClicked;
        event EventHandler? SendJsonCClicked;
        event EventHandler? SendJsonDClicked;
        event EventHandler? SendJsonProxyBClicked;
    }

    /// <summary>ログの重要度レベル</summary>
    public enum LogLevel { Info, Success, Warning, Error }
}
