#nullable enable
using System;
using System.Collections.Generic;
using System.IO;
using System.Threading;
using System.Threading.Tasks;
using Newtonsoft.Json;
using SocketApp.Models;
using SocketApp.Views;

namespace SocketApp.Presenters
{
    public class MainPresenter : IDisposable
    {
        private readonly IMainView _view;

        private readonly SocketClient _clientA;
        private readonly SocketClient _clientB;
        private readonly SocketClient _clientC;
        private readonly SocketClient _clientD;
        private readonly ProxyServerA _proxyServerA;
        private readonly SocketClient _proxyClientB;

        private readonly ScenarioRunner _scenarioRunner;

        private bool _autoDriveActive = false;
        private ClientCResponseConfig _clientCResponseConfig = new ClientCResponseConfig();
        private CancellationTokenSource? _scenarioCts;
        private readonly FileLogger _fileLogger;

        public MainPresenter(IMainView view)
        {
            _view = view;

            var settings = AppConfig.Load();

            _clientA = new SocketClient("ClientA", settings.ClientA.Host, settings.ClientA.Port);
            _clientB = new SocketClient("ClientB", settings.ClientB.Host, settings.ClientB.Port);
            _clientC = new SocketClient("ClientC", settings.ClientC.Host, settings.ClientC.Port);
            _clientD = new SocketClient("ClientD", settings.ClientD.Host, settings.ClientD.Port);
            _proxyServerA = new ProxyServerA("ProxyServerA", settings.ProxyServerA.Port);
            _proxyClientB = new SocketClient("ProxyClientB", settings.ProxyClientB.Host, settings.ProxyClientB.Port);

            _scenarioRunner = new ScenarioRunner(_clientA, _clientB, _clientC, _clientD);

            _view.SetViewListItems(settings.ViewList ?? new List<string>());

            // ClientA〜D, ProxyClientB の共通イベント登録
            foreach (var client in new[] { _clientA, _clientB, _clientC, _clientD, _proxyClientB })
            {
                client.LogMessage += OnClientLog;
                client.ConnectionStateChanged += OnConnectionStateChanged;
                client.NotificationReceived += OnNotificationReceived;
            }

            // ProxyServerA のイベント登録（SocketClient とは異なるイベント体系）
            _proxyServerA.LogMessage += (name, msg, isErr) =>
            {
                _view.AppendLog(msg, isErr ? LogLevel.Error : LogLevel.Info);
                WriteToFileLog(msg, isErr ? "ERROR" : "INFO");
            };
            _proxyServerA.ServerStateChanged += (name, running) =>
            {
                _view.SetConnectionStatus("ProxyServerA", running);
                _view.StartProxyAButtonEnabled = !running;
                _view.StopProxyAButtonEnabled  = running;
                _view.SendProxyAButtonEnabled  = running;
                _view.AppendLog(
                    string.Format("[ProxyServerA] {0}", running ? "起動しました" : "停止しました"),
                    running ? LogLevel.Success : LogLevel.Info);
            };
            _proxyServerA.MessageReceived += (name, message) =>
            {
                _view.AppendLog(
                    string.Format("[ProxyServerA] 受信: {0}", message), LogLevel.Warning);
                WriteToFileLog(string.Format("[ProxyServerA] 受信: {0}", message), "INFO");
            };

            _clientC.ServerRequestReceived += OnClientCServerRequestAsync;

            // ProxyClientB の全受信メッセージを履歴表に追加
            _proxyClientB.RawMessageReceived += OnProxyBRawMessageReceived;

            // ProxyClientB: サーバ起点リクエスト受信時に履歴表へ追加
            _proxyClientB.ServerRequestReceived += (clientName, request) =>
            {
                _view.AddProxyBHistoryRow(
                    DateTime.Now.ToString("HH:mm:ss.fff"),
                    request.Method,
                    request.Id.ToString());
                // デフォルト応答を返す
                return Task.FromResult(new JsonRpcResponse { Id = request.Id, Result = "ok" });
            };

            // ProxyClientB: 通知受信時にも履歴表へ追加
            _proxyClientB.NotificationReceived += (clientName, notification) =>
            {
                _view.AddProxyBHistoryRow(
                    DateTime.Now.ToString("HH:mm:ss.fff"),
                    notification.Method,
                    "(通知)");
            };

            _scenarioRunner.LogMessage    += (msg, isErr) =>
            {
                _view.AppendLog(msg, isErr ? LogLevel.Error : LogLevel.Info);
                WriteToFileLog(msg);
            };
            _scenarioRunner.StateChanged  += OnScenarioStateChanged;
            _scenarioRunner.StepStarted   += (id, client, method) =>
            {
                string msg = string.Format("[Scenario] Step '{0}': {1} → {2}", id, client, method);
                _view.AppendLog(msg, LogLevel.Info);
                WriteToFileLog(msg);
            };
            _scenarioRunner.StepCompleted += (id, success) =>
            {
                string msg = string.Format("[Scenario] Step '{0}' {1}", id, success ? "完了" : "失敗");
                _view.AppendLog(msg, success ? LogLevel.Success : LogLevel.Warning);
                WriteToFileLog(msg);
            };

            // 接続・切断（個別）
            _view.ConnectAClicked    += async (s, e) => await ConnectAsync(_clientA);
            _view.DisconnectAClicked += async (s, e) => await DisconnectAsync(_clientA);
            _view.ConnectBClicked    += async (s, e) => await ConnectAsync(_clientB);
            _view.DisconnectBClicked += async (s, e) => await DisconnectAsync(_clientB);
            _view.ConnectCClicked    += async (s, e) => await ConnectAsync(_clientC);
            _view.DisconnectCClicked += async (s, e) => await DisconnectAsync(_clientC);
            _view.ConnectDClicked    += async (s, e) => await ConnectAsync(_clientD);
            _view.DisconnectDClicked += async (s, e) => await DisconnectAsync(_clientD);
            _view.StartProxyAClicked += (s, e) => StartProxyA();
            _view.StopProxyAClicked  += (s, e) => StopProxyA();
            _view.SendProxyAClicked  += (s, e) => SendProxyAMessage();
            _view.ConnectProxyBClicked    += async (s, e) => await ConnectAsync(_proxyClientB);
            _view.DisconnectProxyBClicked += async (s, e) => await DisconnectAsync(_proxyClientB);

            // その他
            _view.StartDriveClicked            += (s, e) => StartDriveAsync();
            _view.EndDriveClicked              += (s, e) => EndDriveAsync();
            _view.Test1Clicked                 += (s, e) => SendTestAsync("test1Func");
            _view.Test2Clicked                 += (s, e) => SendTestAsync("test2Func");
            _view.Test3Clicked                 += (s, e) => SendTest3Async();
            _view.Test4Clicked                 += (s, e) => SendTest4Async();
            _view.Test5Clicked                 += (s, e) => SendTest5Async();
            _view.ClientCSendClicked           += (s, e) => SendClientCAsync();
            _view.ScenarioRunClicked           += async (s, e) => await RunScenarioAsync();
            _view.ScenarioCancelClicked        += (s, e) => CancelScenario();
            _view.ScenarioBrowseClicked        += (s, e) => BrowseScenarioFile();
            _view.ClientCResponseBrowseClicked += (s, e) => BrowseClientCResponseFile();
            _view.FormClosingEvent += async (s, e) =>
            {
                _view.AppendLog("=== SocketApp 終了 ===", LogLevel.Info);
                WriteToFileLog("=== SocketApp 終了 ===", "SYSTEM");
                await DisconnectAllAsync();
            };

            // JSON ファイル送信イベント
            _view.SendJsonAClicked += (s, e) => SendJsonFile("ClientA", _clientA);
            _view.SendJsonBClicked += (s, e) => SendJsonFile("ClientB", _clientB);
            _view.SendJsonCClicked += (s, e) => SendJsonFile("ClientC", _clientC);
            _view.SendJsonDClicked += (s, e) => SendJsonFile("ClientD", _clientD);
            _view.SendJsonProxyBClicked += (s, e) => SendJsonFile("ProxyClientB", _proxyClientB);

            // FileLogger 初期化
            _fileLogger = new FileLogger();
            ((MainForm)_view).SetLogDirectory(_fileLogger.LogDirectory);

            // アプリ起動ログ
            _view.AppendLog("=== SocketApp 起動 ===", LogLevel.Info);
            WriteToFileLog("=== SocketApp 起動 ===", "SYSTEM");

            UpdateButtonStates();
        }

        // -------------------------------------------------------
        // 接続 / 切断
        // -------------------------------------------------------
        private async Task ConnectAsync(SocketClient client)
        {
            SetConnectButtonEnabled(client, false);
            try { await client.ConnectAsync(); }
            catch (Exception ex)
            {
                _view.AppendLog(
                    string.Format("[{0}] 接続エラー: {1}", client.ClientName, ex.Message),
                    LogLevel.Error);
            }
            UpdateButtonStates();
        }

        private async Task DisconnectAsync(SocketClient client)
        {
            try { await client.DisconnectAsync(); }
            catch { }
            UpdateButtonStates();
        }

        private async Task DisconnectAllAsync()
        {
            // SocketClient の切断
            foreach (var c in new[] { _clientA, _clientB, _clientC, _clientD, _proxyClientB })
            {
                try { await c.DisconnectAsync(); }
                catch { }
            }
            // ProxyServerA の停止
            _proxyServerA.Stop();
            UpdateButtonStates();
        }

        // -------------------------------------------------------
        // 自動運転（ClientA）- 送信後すぐボタン再有効化
        // -------------------------------------------------------
        private void StartDriveAsync()
        {
            if (!_clientA.IsConnected) { NotifyNotConnected("ClientA"); return; }
            _autoDriveActive = true;
            UpdateButtonStates();
            SendInBackground("ClientA", _clientA, "StartCommand");
        }

        private void EndDriveAsync()
        {
            if (!_clientA.IsConnected) { NotifyNotConnected("ClientA"); return; }
            _autoDriveActive = false;
            UpdateButtonStates();
            SendInBackground("ClientA", _clientA, "EndCommand");
        }

        // -------------------------------------------------------
        // テスト1 / テスト2（ClientB）- 送信後すぐボタン再有効化
        // -------------------------------------------------------
        private void SendTestAsync(string method)
        {
            if (!_clientB.IsConnected) { NotifyNotConnected("ClientB"); return; }
            SendInBackground("ClientB", _clientB, method);
        }

        // -------------------------------------------------------
        // テスト3（ClientB）- 送信後すぐボタン再有効化
        // -------------------------------------------------------
        private void SendTest3Async()
        {
            if (!_clientB.IsConnected) { NotifyNotConnected("ClientB"); return; }
            string selectedView = _view.SelectedViewListItem;
            if (string.IsNullOrEmpty(selectedView))
            {
                _view.ShowWarning("viewList の値を選択してください。", "選択エラー");
                return;
            }
            var parameters = new { viewList = selectedView };
            SendInBackground("ClientB", _clientB, "test3Func", parameters);
        }

        // -------------------------------------------------------
        // テスト4（ClientD）- 送信後すぐボタン再有効化
        // -------------------------------------------------------
        private void SendTest4Async()
        {
            if (!_clientD.IsConnected) { NotifyNotConnected("ClientD"); return; }
            var parameters = new Test4Params
            {
                Info1 = _view.Test4Info1,
                Info2 = _view.Test4Info2,
                Info3 = _view.Test4Info3,
                Info4 = _view.Test4Info4,
            };
            _view.AppendLog(
                string.Format("[ClientD] test4Func params: {0}", JsonConvert.SerializeObject(parameters)),
                LogLevel.Info);
            SendInBackground("ClientD", _clientD, "test4Func", parameters);
        }

        // -------------------------------------------------------
        // テスト5（ClientD）
        // -------------------------------------------------------
        // -------------------------------------------------------
        // テスト5（ClientD）- 送信後すぐボタン再有効化、応答はバックグラウンドで解析
        // -------------------------------------------------------
        private void SendTest5Async()
        {
            if (!_clientD.IsConnected) { NotifyNotConnected("ClientD"); return; }
            _ = Task.Run(async () =>
            {
                try
                {
                    var response = await _clientD.SendRequestAsync("test5Func");
                    if (response.Error != null)
                    {
                        _view.AppendLog(
                            string.Format("[ClientD] test5Func → エラー応答 code={0} msg={1}",
                                response.Error.Code, response.Error.Message),
                            LogLevel.Error);
                    }
                    else if (response.Result != null)
                    {
                        try
                        {
                            var resultObj = JsonConvert.DeserializeObject<Test5Result>(
                                JsonConvert.SerializeObject(response.Result));
                            if (resultObj != null)
                            {
                                _view.AppendLog(
                                    string.Format("[ClientD] test5Func → 成功: {0}", resultObj.ToString()),
                                    LogLevel.Success);
                                if (!resultObj.Val3.HasValue || !resultObj.Val4.HasValue)
                                {
                                    _view.AppendLog(
                                        string.Format("[ClientD] test5Func 注意: val3={0}, val4={1}",
                                            resultObj.Val3.HasValue ? resultObj.Val3.Value.ToString() : "(なし)",
                                            resultObj.Val4.HasValue ? resultObj.Val4.Value.ToString() : "(なし)"),
                                        LogLevel.Warning);
                                }
                            }
                            else
                            {
                                _view.AppendLog("[ClientD] test5Func → result の解析結果が null", LogLevel.Warning);
                            }
                        }
                        catch (Exception parseEx)
                        {
                            _view.AppendLog(
                                string.Format("[ClientD] test5Func → result 解析失敗 ({0}), 生データ: {1}",
                                    parseEx.Message, JsonConvert.SerializeObject(response.Result)),
                                LogLevel.Warning);
                        }
                    }
                    else
                    {
                        _view.AppendLog("[ClientD] test5Func → 成功: result=(null)", LogLevel.Success);
                    }
                }
                catch (Exception ex)
                {
                    _view.AppendLog(string.Format("[ClientD] test5Func エラー: {0}", ex.Message), LogLevel.Error);
                }
            });
        }

        // -------------------------------------------------------
        // ClientC 手動送信 - 送信後すぐボタン再有効化
        // -------------------------------------------------------
        private void SendClientCAsync()
        {
            string method = _view.ClientCMethod;
            if (string.IsNullOrEmpty(method)) { _view.ShowWarning("method を入力してください。", "入力エラー"); return; }
            if (!_clientC.IsConnected) { NotifyNotConnected("ClientC"); return; }
            SendInBackground("ClientC", _clientC, method);
        }

        // -------------------------------------------------------
        // ClientC 自動応答
        // -------------------------------------------------------
        private async Task<JsonRpcResponse> OnClientCServerRequestAsync(string clientName, JsonRpcRequest request)
        {
            _view.AppendLog(
                string.Format("[ClientC] サーバ起点リクエスト受信: method={0}", request.Method),
                LogLevel.Warning);

            if (_clientCResponseConfig.Responses.TryGetValue(request.Method, out var def))
            {
                if (def.DelayMs > 0) await Task.Delay(def.DelayMs);
                if (def.IsError)
                {
                    _view.AppendLog(string.Format("[ClientC] エラー応答: code={0}", def.ErrorCode), LogLevel.Warning);
                    return new JsonRpcResponse
                    {
                        Id = request.Id,
                        Error = new JsonRpcError { Code = def.ErrorCode, Message = def.ErrorMessage }
                    };
                }
                else
                {
                    _view.AppendLog(
                        string.Format("[ClientC] 正常応答: result={0}", JsonConvert.SerializeObject(def.Result)),
                        LogLevel.Success);
                    return new JsonRpcResponse { Id = request.Id, Result = def.Result };
                }
            }

            _view.AppendLog(
                string.Format("[ClientC] 応答定義なし（デフォルト応答）: method={0}", request.Method),
                LogLevel.Warning);
            return new JsonRpcResponse { Id = request.Id, Result = "ok" };
        }

        // -------------------------------------------------------
        // ProxyServerA（CR+LF区切り汎用TCPサーバ）操作
        // -------------------------------------------------------

        /// <summary>ProxyServerA を起動する</summary>
        private void StartProxyA()
        {
            if (_proxyServerA.IsRunning)
            {
                _view.ShowWarning("既に起動中です", "起動失敗");
                return;
            }

            _proxyServerA.Start();
            UpdateButtonStates();
        }

        /// <summary>ProxyServerA を停止する</summary>
        private void StopProxyA()
        {
            if (!_proxyServerA.IsRunning)
            {
                _view.ShowWarning("既に停止中です", "停止失敗");
                return;
            }

            _proxyServerA.Stop();
            UpdateButtonStates();
        }

        /// <summary>ProxyServerA 経由で接続中の全クライアントにテキストメッセージを送信する</summary>
        private void SendProxyAMessage()
        {
            if (!_proxyServerA.IsRunning)
            {
                _view.ShowWarning("ProxyServerA が起動していません", "送信失敗");
                return;
            }

            string message = _view.ProxyASendText;
            if (string.IsNullOrWhiteSpace(message))
            {
                _view.ShowWarning("送信メッセージを入力してください", "入力エラー");
                return;
            }

            _ = Task.Run(async () =>
            {
                try
                {
                    await _proxyServerA.SendToAllAsync(message);
                    _view.AppendLog(
                        string.Format("[ProxyServerA] テキスト送信完了: {0}", message),
                        LogLevel.Success);
                    WriteToFileLog(
                        string.Format("[ProxyServerA] テキスト送信: {0}", message), "INFO");
                }
                catch (Exception ex)
                {
                    _view.AppendLog(
                        string.Format("[ProxyServerA] 送信エラー: {0}", ex.Message),
                        LogLevel.Error);
                    WriteToFileLog(
                        string.Format("[ProxyServerA] 送信エラー: {0}", ex.Message), "ERROR");
                }
            });
        }

        // -------------------------------------------------------
        // シナリオ実行
        // -------------------------------------------------------
        private async Task RunScenarioAsync()
        {
            string scenarioPath   = _view.ScenarioFilePath;
            string ccResponsePath = _view.ClientCResponseFilePath;

            if (string.IsNullOrEmpty(scenarioPath))
            {
                _view.ShowWarning("シナリオファイルを選択してください。", "ファイル未選択");
                return;
            }

            if (!string.IsNullOrEmpty(ccResponsePath) && File.Exists(ccResponsePath))
            {
                try
                {
                    _clientCResponseConfig = ScenarioRunner.LoadClientCResponses(ccResponsePath);
                    _view.AppendLog(
                        string.Format("[Scenario] ClientC応答定義を読み込みました: {0}", ccResponsePath),
                        LogLevel.Info);
                }
                catch (Exception ex)
                {
                    _view.ShowWarning(
                        string.Format("ClientC応答定義ファイルの読み込みに失敗しました。\n{0}", ex.Message),
                        "読み込みエラー");
                }
            }

            ScenarioConfig scenario;
            try { scenario = ScenarioRunner.LoadScenario(scenarioPath); }
            catch (Exception ex)
            {
                _view.ShowError(
                    string.Format("シナリオファイルの読み込みに失敗しました。\n{0}", ex.Message),
                    "読み込みエラー");
                return;
            }

            _scenarioCts = new CancellationTokenSource();
            await _scenarioRunner.RunAsync(scenario, scenarioPath, _scenarioCts.Token);
        }

        private void CancelScenario() => _scenarioCts?.Cancel();

        private void BrowseScenarioFile()
        {
            string? path = _view.ShowOpenFileDialog(
                "シナリオファイルを選択", "JSONファイル (*.json)|*.json|全てのファイル (*.*)|*.*");
            if (path != null) ((MainForm)_view).SetScenarioFilePath(path);
        }

        private void BrowseClientCResponseFile()
        {
            string? path = _view.ShowOpenFileDialog(
                "ClientC応答定義ファイルを選択", "JSONファイル (*.json)|*.json|全てのファイル (*.*)|*.*");
            if (path != null) ((MainForm)_view).SetClientCResponseFilePath(path);
        }

        // -------------------------------------------------------
        // シナリオ状態変化
        // -------------------------------------------------------
        private void OnScenarioStateChanged(ScenarioState state)
        {
            string statusText;
            switch (state)
            {
                case ScenarioState.Running:   statusText = "● 実行中..."; break;
                case ScenarioState.Completed: statusText = "✔ 完了";      break;
                case ScenarioState.Aborted:   statusText = "✖ 異常終了";  break;
                default:                      statusText = "待機中";       break;
            }
            _view.SetScenarioStatus(statusText);
            _view.ScenarioRunButtonEnabled    = state != ScenarioState.Running;
            _view.ScenarioCancelButtonEnabled = state == ScenarioState.Running;
        }

        // -------------------------------------------------------
        // Model イベントハンドラ
        // -------------------------------------------------------
        private void OnConnectionStateChanged(string clientName, bool isConnected)
        {
            _view.SetConnectionStatus(clientName, isConnected);
            _view.AppendLog(
                string.Format("[{0}] {1}", clientName, isConnected ? "接続しました" : "切断しました"),
                isConnected ? LogLevel.Success : LogLevel.Info);
            UpdateButtonStates();
        }

        private void OnClientLog(string clientName, string message)
        {
            _view.AppendLog(message, LogLevel.Info);
            WriteToFileLog(message);
        }

        /// <summary>
        /// ProxyClientB の全受信メッセージを解析し、受信履歴表に追加する。
        /// JSON 解析に失敗した場合も行を追加する（method="(parse error)", id="N/A"）。
        /// </summary>
        private void OnProxyBRawMessageReceived(string clientName, string rawJson)
        {
            string time = DateTime.Now.ToString("HH:mm:ss.fff");
            string method = "(unknown)";
            string id = "N/A";

            try
            {
                var jobj = Newtonsoft.Json.Linq.JObject.Parse(rawJson);

                // method の取得
                var methodToken = jobj["method"];
                if (methodToken != null)
                    method = methodToken.ToString();

                // id の取得
                var idToken = jobj["id"];
                if (idToken != null && idToken.Type != Newtonsoft.Json.Linq.JTokenType.Null)
                    id = idToken.ToString();
            }
            catch
            {
                method = "(parse error)";
            }

            _view.AddProxyBHistoryRow(time, method, id);
        }

        /// <summary>ファイルログへの書き込み（チェックボックス連動）</summary>
        /// <param name="message">メッセージ</param>
        /// <param name="level">ログレベル文字列（INFO/SUCCESS/WARNING/ERROR/SYSTEM）</param>
        private void WriteToFileLog(string message, string level = "INFO")
        {
            _fileLogger.Enabled = _view.FileLogEnabled;
            _fileLogger.Write(message, level);
        }

        private void OnNotificationReceived(string clientName, JsonRpcNotification notification)
        {
            string paramsStr = notification.Params != null
                ? JsonConvert.SerializeObject(notification.Params) : "";
            _view.AppendLog(
                string.Format("[{0}] 通知受信 method={1} params={2}",
                    clientName, notification.Method, paramsStr),
                LogLevel.Warning);

            if (notification.Method == "ErrorInfo" && _autoDriveActive)
                _view.ShowWarning(
                    string.Format("サーバからエラー通知を受信しました。\n\n{0}", paramsStr),
                    "エラー通知 (ErrorInfo)");
        }

        // -------------------------------------------------------
        // レスポンス処理
        // -------------------------------------------------------
        private void HandleResponse(string clientName, string method, JsonRpcResponse response)
        {
            if (response.Error != null)
            {
                string msg = string.Format("[{0}] {1} → エラー応答 code={2} msg={3}",
                    clientName, method, response.Error.Code, response.Error.Message);
                _view.AppendLog(msg, LogLevel.Error);
                WriteToFileLog(msg, "ERROR");
            }
            else
            {
                string resultStr = response.Result != null
                    ? JsonConvert.SerializeObject(response.Result) : "(null)";
                string msg = string.Format("[{0}] {1} → 成功: {2}", clientName, method, resultStr);
                _view.AppendLog(msg, LogLevel.Success);
                WriteToFileLog(msg, "SUCCESS");
            }
        }

        // -------------------------------------------------------
        // 共通: バックグラウンド送信ヘルパー
        // 送信とログ出力をバックグラウンドで行い、呼び出し元は即座に戻る
        // -------------------------------------------------------
        private void SendInBackground(string clientName, SocketClient client, string method, object? parameters = null)
        {
            _ = Task.Run(async () =>
            {
                try
                {
                    var response = await client.SendRequestAsync(method, parameters);
                    HandleResponse(clientName, method, response);
                }
                catch (Exception ex)
                {
                    _view.AppendLog(
                        string.Format("[{0}] {1} エラー: {2}", clientName, method, ex.Message),
                        LogLevel.Error);
                }
            });
        }

        // -------------------------------------------------------
        // 共通: JSON ファイル生送信
        //
        // ファイル選択ダイアログを開き、選択されたファイルの内容を
        // バリデーションせずにそのまま送信する。
        // 異常な JSON や不正データの送信テストに使用できる。
        //
        // ・ファイルの中身は一切チェックしない（壊れた JSON でも送信可能）
        // ・末尾に EOT (\x04) を自動付与して送信する
        // ・応答はバックグラウンドで SocketClient の受信ループが処理する
        // -------------------------------------------------------
        private void SendJsonFile(string clientName, SocketClient client)
        {
            // 接続チェック
            if (!client.IsConnected)
            {
                NotifyNotConnected(clientName);
                return;
            }

            // ファイル選択ダイアログ
            string? filePath = _view.ShowOpenFileDialog(
                string.Format("{0} に送信するファイルを選択", clientName),
                "JSONファイル (*.json)|*.json|全てのファイル (*.*)|*.*");

            if (filePath == null) return;  // キャンセル時

            // ファイル読み込み → そのまま送信
            try
            {
                string rawContent = File.ReadAllText(filePath);

                // 送信内容をログに記録
                string logMsg = string.Format(
                    "[{0}] ファイル生送信: {1} ({2} bytes)",
                    clientName, filePath, rawContent.Length);
                _view.AppendLog(logMsg, LogLevel.Info);
                WriteToFileLog(logMsg);

                // 送信内容のプレビューをログに出力（長い場合は先頭200文字まで）
                string preview = rawContent.Length > 200
                    ? rawContent.Substring(0, 200) + "..."
                    : rawContent;
                _view.AppendLog(
                    string.Format("[{0}] 送信内容: {1}", clientName, preview),
                    LogLevel.Info);

                // バックグラウンドで生データ送信
                _ = Task.Run(async () =>
                {
                    try
                    {
                        await client.SendRawAsync(rawContent);
                        _view.AppendLog(
                            string.Format("[{0}] ファイル送信完了", clientName),
                            LogLevel.Success);
                        WriteToFileLog(
                            string.Format("[{0}] ファイル送信完了: {1}", clientName, filePath),
                            "SUCCESS");
                    }
                    catch (Exception ex)
                    {
                        _view.AppendLog(
                            string.Format("[{0}] ファイル送信エラー: {1}", clientName, ex.Message),
                            LogLevel.Error);
                        WriteToFileLog(
                            string.Format("[{0}] ファイル送信エラー: {1}", clientName, ex.Message),
                            "ERROR");
                    }
                });
            }
            catch (Exception ex)
            {
                _view.ShowError(
                    string.Format("ファイルの読み込みに失敗しました。\n\n{0}", ex.Message),
                    "ファイル読み込みエラー");
            }
        }

        // -------------------------------------------------------
        // ボタン状態管理
        // -------------------------------------------------------
        private void UpdateButtonStates()
        {
            _view.ConnectAButtonEnabled    = !_clientA.IsConnected;
            _view.DisconnectAButtonEnabled =  _clientA.IsConnected;
            _view.ConnectBButtonEnabled    = !_clientB.IsConnected;
            _view.DisconnectBButtonEnabled =  _clientB.IsConnected;
            _view.ConnectCButtonEnabled    = !_clientC.IsConnected;
            _view.DisconnectCButtonEnabled =  _clientC.IsConnected;
            _view.ConnectDButtonEnabled    = !_clientD.IsConnected;
            _view.DisconnectDButtonEnabled =  _clientD.IsConnected;
            _view.StartProxyAButtonEnabled  = !_proxyServerA.IsRunning;
            _view.StopProxyAButtonEnabled   =  _proxyServerA.IsRunning;
            _view.SendProxyAButtonEnabled   =  _proxyServerA.IsRunning;
            _view.ConnectProxyBButtonEnabled    = !_proxyClientB.IsConnected;
            _view.DisconnectProxyBButtonEnabled =  _proxyClientB.IsConnected;

            _view.StartDriveButtonEnabled     = _clientA.IsConnected && !_autoDriveActive;
            _view.EndDriveButtonEnabled       = _clientA.IsConnected && _autoDriveActive;
            _view.Test1ButtonEnabled          = _clientB.IsConnected;
            _view.Test2ButtonEnabled          = _clientB.IsConnected;
            _view.Test3ButtonEnabled          = _clientB.IsConnected;
            _view.Test4ButtonEnabled          = _clientD.IsConnected;
            _view.Test5ButtonEnabled          = _clientD.IsConnected;
            _view.ClientCSendButtonEnabled    = _clientC.IsConnected;
            _view.ScenarioRunButtonEnabled    = _scenarioRunner.State != ScenarioState.Running;
            _view.ScenarioCancelButtonEnabled = _scenarioRunner.State == ScenarioState.Running;
        }

        private void SetConnectButtonEnabled(SocketClient client, bool enabled)
        {
            if      (client == _clientA)      _view.ConnectAButtonEnabled      = enabled;
            else if (client == _clientB)      _view.ConnectBButtonEnabled      = enabled;
            else if (client == _clientC)      _view.ConnectCButtonEnabled      = enabled;
            else if (client == _clientD)      _view.ConnectDButtonEnabled      = enabled;
            else if (client == _proxyClientB) _view.ConnectProxyBButtonEnabled = enabled;
        }

        private void NotifyNotConnected(string clientName) =>
            _view.ShowWarning(
                string.Format("{0} が未接続です。先に接続してください。", clientName), "未接続");

        public void Dispose()
        {
            _fileLogger.Dispose();
            _clientA.Dispose();
            _clientB.Dispose();
            _clientC.Dispose();
            _clientD.Dispose();
            _proxyServerA.Dispose();
            _proxyClientB.Dispose();
        }
