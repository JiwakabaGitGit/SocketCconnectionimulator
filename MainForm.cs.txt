        public void SetConnectionStatus(string clientName, bool isConnected)
        {
            UIInvoke(() =>
            {
                Label lamp = clientName == "ClientA" ? lblStatusA
                           : clientName == "ClientB" ? lblStatusB
                           : clientName == "ClientC" ? lblStatusC
                           : clientName == "ClientD" ? lblStatusD
                           : clientName == "ProxyClientA" ? lblStatusProxyA
                           : clientName == "ProxyClientB" ? lblStatusProxyB
                           : lblStatusD;
                lamp.BackColor = isConnected ? Color.LimeGreen : Color.Gray;

                Label state = clientName == "ClientA" ? lblStatusLabelA
                            : clientName == "ClientB" ? lblStatusLabelB
                            : clientName == "ClientC" ? lblStatusLabelC
                            : clientName == "ClientD" ? lblStatusLabelD
                            : clientName == "ProxyClientA" ? lblStatusLabelProxyA
                            : clientName == "ProxyClientB" ? lblStatusLabelProxyB
                            : lblStatusLabelD;
                state.Text      = isConnected ? "●接続中" : "●切断中";
                state.ForeColor = isConnected ? Color.Green : Color.Gray;
            });
        }

        public void AppendLog(string message, LogLevel level)
        {
            UIInvoke(() =>
            {
                Color color = level == LogLevel.Success ? Color.LightGreen
                            : level == LogLevel.Warning ? Color.Yellow
                            : level == LogLevel.Error   ? Color.OrangeRed

        // ---- IMainView: ProxyClientB 受信履歴表 ----
        private int _proxyBHistoryNo = 0;

        /// <summary>
        /// ProxyClientB の受信履歴を DataGridView に1行追加する。
        /// </summary>
        /// <param name="time">受信時刻（書式済み文字列）</param>
        /// <param name="method">JSON-RPC の method 名</param>
        /// <param name="id">JSON-RPC の id（文字列表現）</param>
        public void AddProxyBHistoryRow(string time, string method, string id)
        {
            UIInvoke(() =>
            {
                _proxyBHistoryNo++;
                dgvProxyBHistory.Rows.Add(_proxyBHistoryNo.ToString(), time, method, id);

                // 最終行が見えるよう自動スクロール
                if (dgvProxyBHistory.Rows.Count > 0)
                    dgvProxyBHistory.FirstDisplayedScrollingRowIndex = dgvProxyBHistory.Rows.Count - 1;
            });
        }

        /// <summary>
        /// スレッドセーフな UI 更新ヘルパー。
        /// ・フォームが既に破棄されている場合は何もしない（アプリ終了時の例外を防ぐ）
        /// ・フォームのハンドルが未作成（Application.Run 前）の場合は直接実行する
        /// ・別スレッドからの呼び出しの場合は Invoke で UI スレッドにマーシャリングする
        /// </summary>
        private void UIInvoke(Action action)
        {
            // フォームが既に破棄されている場合は何もしない
            if (IsDisposed) return;

            try
            {
                // ハンドル未作成（フォーム表示前）→ まだ UI スレッドなので直接実行
                if (!IsHandleCreated)
                {
                    action();
                    return;
                }

                // 別スレッドからの呼び出し → Invoke で UI スレッドへマーシャリング
                if (InvokeRequired)
                    Invoke(action);
                else
                    action();
            }
            catch (ObjectDisposedException)
            {
                // Dispose 直後のタイミングで Invoke が呼ばれた場合は無視
            }
            catch (InvalidOperationException)
            {
                // ハンドルが無効な場合も無視
            }
        }
